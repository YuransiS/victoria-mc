      <div className="flex items-baseline gap-2">
        <p className="text-3xl font-black font-headline tracking-tighter">{value}</p>

}

function InfoRow({ 
  icon, 
  label, 
  value, 
  color = "text-white/50", 
  isCopyable = false,
  isTelegram = false,
  italic = false
}: { 
  icon: React.ReactNode, 
  label: string, 
  value: string | number, 
  color?: string,
  isCopyable?: boolean,
  isTelegram?: boolean,
  italic?: boolean
}) {
  const [copied, setCopied] = useState(false);

  const handleCopy = () => {
    if (value && value !== 'вЂ”') {
      navigator.clipboard.writeText(value.toString());
      setCopied(true);
      setTimeout(() => setCopied(false), 2000);
    }
  };

  const tgLink = isTelegram && value && value !== 'вЂ”' 
    ? `https://telegram.me/${value.toString().replace('@', '').trim()}` 
    : null;

  return (
    <div className="flex items-start gap-4 group/info">
      <div className="text-white/10 mt-1">{icon}</div>
      <div className="flex-1 min-w-0">
        <p className="text-[7px] uppercase font-black text-white/10 tracking-[0.2em] mb-1">{label}</p>
        <div className="flex items-center gap-2">
          {tgLink ? (
            <a href={tgLink} target="_blank" rel="noopener noreferrer" className={`text-[11px] font-bold ${color} hover:text-[#D4AF37] transition-colors truncate`}>
              {value}
            </a>
          ) : (
            <p className={`text-[11px] font-bold ${color} truncate ${italic ? 'italic font-medium' : ''}`}>{value}</p>
          )}

          {isCopyable && value && value !== 'вЂ”' && (
            <button onClick={handleCopy} className="opacity-0 group-hover/info:opacity-100 transition-opacity">
              {copied ? <Check size={10} className="text-[#D4AF37]" /> : <Copy size={10} className="text-white/10" />}
            </button>
          )}
        </div>
      </div>
    </div>
  );
}

  return (

      <select

      >
        {options.map(o => (
          <option key={o.v} value={o.v} className="bg-[#0a0a0a]">{o.l}</option>
        ))}
      </select>
    </div>
  );
}
